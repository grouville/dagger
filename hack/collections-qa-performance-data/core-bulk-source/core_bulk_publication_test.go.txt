func TestCoreBulkPublicationMetadataOwnership(t *testing.T) {
	ctx := t.Context()
	cache, err := dagql.NewCache(ctx, "", nil, nil)
	require.NoError(t, err)
	t.Cleanup(func() { require.NoError(t, cache.Close(context.Background())) })
	ctx = dagql.ContextWithCache(ctx, cache)
	ctx = engine.ContextWithClientMetadata(ctx, &engine.ClientMetadata{ClientID: "bulk-producer", SessionID: "bulk-producer"})
	server := &currentTypeDefsTestServer{}
	root := core.NewRoot(server)
	ctx = core.ContextWithQuery(ctx, root)
	base, err := NewCoreSchemaBase(ctx, server)
	require.NoError(t, err)
	dag, err := base.Fork(ctx, root, "v1.0.0")
	require.NoError(t, err)
	server.dag = dag
	var td dagql.ObjectResult[*core.TypeDef]
	require.NoError(t, dag.Select(ctx, dag.Root(), &td, dagql.Selector{Field: "typeDef"}, dagql.Selector{Field: "withKind", Args: []dagql.NamedInput{{Name: "kind", Value: core.TypeDefKindString}}}))
	typeID, err := core.ResultIDInput(td)
	require.NoError(t, err)
	ids := dagql.ArrayInput[core.FunctionID]{}
	for _, name := range []string{"first", "second"} {
		var fn dagql.ObjectResult[*core.Function]
		require.NoError(t, dag.Select(ctx, dag.Root(), &fn,
			dagql.Selector{Field: "function", Args: []dagql.NamedInput{{Name: "name", Value: dagql.String(name)}, {Name: "returnType", Value: typeID}}},
			dagql.Selector{Field: "withDescription", Args: []dagql.NamedInput{{Name: "description", Value: dagql.String("retained metadata")}}},
			dagql.Selector{Field: "withArg", Args: []dagql.NamedInput{{Name: "name", Value: dagql.String("value")}, {Name: "typeDef", Value: typeID}, {Name: "defaultValue", Value: core.JSON(`"default"`)}}},
		))
		id, err := core.ResultIDInput(fn)
		require.NoError(t, err)
		ids = append(ids, id)
	}
	var obj dagql.ObjectResult[*core.ObjectTypeDef]
	require.NoError(t, dag.Select(ctx, dag.Root(), &obj, dagql.Selector{Field: "__objectTypeDef", Args: []dagql.NamedInput{{Name: "name", Value: dagql.String("Example")}}}))
	sequential := obj
	for _, id := range ids {
		require.NoError(t, dag.Select(ctx, sequential, &sequential, dagql.Selector{Field: "__withFunction", Args: []dagql.NamedInput{{Name: "function", Value: id}}}))
	}
	var bulk dagql.ObjectResult[*core.ObjectTypeDef]
	require.NoError(t, dag.Select(ctx, obj, &bulk, dagql.Selector{Field: "__withFunctions", Args: []dagql.NamedInput{{Name: "functions", Value: ids}}}))
	require.Empty(t, obj.Self().Functions)
	require.Len(t, bulk.Self().Functions, 2)
	for i, fn := range bulk.Self().Functions {
		want, err := sequential.Self().Functions[i].ID()
		require.NoError(t, err)
		got, err := fn.ID()
		require.NoError(t, err)
		require.Equal(t, want.EngineResultID(), got.EngineResultID())
		require.Equal(t, "retained metadata", fn.Self().Description)
		require.JSONEq(t, `"default"`, string(fn.Self().Args[0].Self().DefaultValue))
	}
	// A final result must keep its function/argument/type graph alive after the
	// input session ends, using the ordinary Dagger attachment/ownership path.
	consumer := engine.ContextWithClientMetadata(ctx, &engine.ClientMetadata{ClientID: "bulk-consumer", SessionID: "bulk-consumer"})
	bulkID, err := bulk.ID()
	require.NoError(t, err)
	held, err := dag.Load(consumer, bulkID)
	require.NoError(t, err)
	require.NoError(t, cache.ReleaseSession(ctx, "bulk-producer"))
	typed, ok := held.(dagql.ObjectResult[*core.ObjectTypeDef])
	require.True(t, ok)
	for _, fn := range typed.Self().Functions {
		id, err := fn.ID()
		require.NoError(t, err)
		loaded, err := dag.Load(consumer, id)
		require.NoError(t, err)
		require.Equal(t, "Function", loaded.Type().Name())
		argID, err := fn.Self().Args[0].ID()
		require.NoError(t, err)
		_, err = dag.Load(consumer, argID)
		require.NoError(t, err)
	}
}

func TestCoreBulkPublicationUsesOneFinalUpdate(t *testing.T) {
	for _, n := range []int{0, 1, 10} {
		for _, isInterface := range []bool{false, true} {
			t.Run(fmt.Sprintf("functions=%d/interface=%t", n, isInterface), func(t *testing.T) {
				ctx := t.Context()
				cache, err := dagql.NewCache(ctx, "", nil, nil)
				require.NoError(t, err)
				t.Cleanup(func() { require.NoError(t, cache.Close(context.Background())) })
				ctx = dagql.ContextWithCache(ctx, cache)
				ctx = engine.ContextWithClientMetadata(ctx, &engine.ClientMetadata{ClientID: "core-bulk", SessionID: "core-bulk"})
				server := &currentTypeDefsTestServer{}
				root := core.NewRoot(server)
				ctx = core.ContextWithQuery(ctx, root)
				base, err := NewCoreSchemaBase(ctx, server)
				require.NoError(t, err)
				dag, err := base.Fork(ctx, root, "v1.0.0")
				require.NoError(t, err)
				server.dag = dag
				var single, bulk atomic.Int64
				dag.Around(func(ctx context.Context, req *dagql.CallRequest) (context.Context, func(dagql.AnyResult, bool, *error)) {
					if req.Field == "__withFunction" {
						single.Add(1)
					}
					if req.Field == "__withFunctions" {
						bulk.Add(1)
					}
					return ctx, dagql.NoopDone
				})
				shape := &introspection.Type{Name: "BulkFixture", Description: "Declared docs", Fields: []*introspection.Field{{Name: "id"}}}
				for i := 0; i < n; i++ {
					shape.Fields = append(shape.Fields, &introspection.Field{Name: fmt.Sprintf("field%d", i), Description: "Function docs", TypeRef: &introspection.TypeRef{Kind: introspection.TypeKindNonNull, OfType: &introspection.TypeRef{Kind: introspection.TypeKindScalar, Name: "String"}}})
				}
				var td dagql.ObjectResult[*core.TypeDef]
				var ok bool
				if isInterface {
					td, ok, err = buildCoreObjectLikeTypeDef[*core.InterfaceTypeDef](ctx, dag, shape, "__interfaceTypeDef", "__withInterfaceTypeDef", "interfaceTypeDef")
				} else {
					td, ok, err = buildCoreObjectLikeTypeDef[*core.ObjectTypeDef](ctx, dag, shape, "__objectTypeDef", "__withObjectTypeDef", "objectTypeDef")
				}
				require.NoError(t, err)
				require.True(t, ok)
				var functions dagql.ObjectResultArray[*core.Function]
				if isInterface {
					functions = td.Self().AsInterface.Value.Self().Functions
				} else {
					functions = td.Self().AsObject.Value.Self().Functions
				}
				require.Len(t, functions, n)
				for i, fn := range functions {
					require.Equal(t, fmt.Sprintf("field%d", i), fn.Self().Name)
					require.Equal(t, "Function docs", fn.Self().Description)
					require.Equal(t, core.TypeDefKindString, fn.Self().ReturnType.Self().Kind)
					require.False(t, fn.Self().ReturnType.Self().Optional)
				}
				if n > 1 {
					require.EqualValues(t, 1, bulk.Load())
					require.Zero(t, single.Load())
				} else {
					require.Zero(t, bulk.Load())
					require.EqualValues(t, n, single.Load())
				}
			})
		}
	}
}

func BenchmarkCoreTypeDefPublication(b *testing.B) {
	// Each sample constructs fresh authoritative metadata; setup/teardown and
	// installed GraphQL schema construction are outside this measured region.
	for b.Loop() {
		b.StopTimer()
		ctx := b.Context()
		cache, err := dagql.NewCache(ctx, "", nil, nil)
		if err != nil {
			b.Fatal(err)
		}
		ctx = dagql.ContextWithCache(ctx, cache)
		ctx = engine.ContextWithClientMetadata(ctx, &engine.ClientMetadata{ClientID: "core-bulk-bench", SessionID: "core-bulk-bench"})
		server := &currentTypeDefsTestServer{}
		base, err := NewCoreSchemaBase(ctx, server)
		if err != nil {
			b.Fatal(err)
		}
		b.StartTimer()
		defs, err := base.CoreMod("v1.0.0").TypeDefs(ctx, nil)
		b.StopTimer()
		if err != nil {
			b.Fatal(err)
		}
		if len(defs) == 0 {
			b.Fatal("empty core metadata")
		}
		if err := cache.Close(context.Background()); err != nil {
			b.Fatal(err)
		}
		b.StartTimer()
	}
}
