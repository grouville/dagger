"""Reduce completed pruning evidence to fixed classes, counts and numeric timings."""
from pathlib import Path
import hashlib,json,statistics
P=Path(__file__).resolve().parent;R=P/'results-v1';O=P/'evidence-v1'
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(p,v):p.write_text(json.dumps(v,indent=2)+'\n')
def union(xs):
 merged=[]
 for a,b in sorted(xs):
  if merged and a<=merged[-1][1]:merged[-1][1]=max(merged[-1][1],b)
  else:merged.append([a,b])
 return sum(b-a for a,b in merged)/1e6
def profile(p):
 with p.open()as f:h=json.loads(next(f));ops=[json.loads(l)for l in f if l.strip()]
 ops=[o for o in ops if o.get('e')=='op'];by={o['id']:o for o in ops}
 def name(o):return h['strings'][o.get('c',0)]
 def children(root):
  result=[]
  for o in ops:
   current=o;seen=set()
   while current and current['id']not in seen:
    if current['id']==root['id']:result.append(o);break
    seen.add(current['id']);current=by.get(current.get('p'))
  return result
 def stats(xs):return {'count':len(xs),'inclusive_sum_ms':sum(o['d']-o['s']for o in xs)/1e6,'union_ms':union((o['s'],o['d'])for o in xs),'max_ms':max([o['d']-o['s']for o in xs],default=0)/1e6,'cache_hits':sum(o.get('o')in ('hit','cache_hit')for o in xs),'errors':sum(o.get('o')=='error'for o in xs)}
 phases=['Workspace.artifacts','Artifacts.__itemsJSON','Address.container','Query.typeDef','TypeDef.withOptional','Module._implementationScoped','backend:Query.backend','backend:Backend.goTestBase','go:Go.modules','go:GoModule.tests','session.query','exec.processRun']
 def aggregates(xs):return [{'kind':kind,'class':cl,**stats([o for o in xs if name(o)==cl and o['k']==kind])}for cl in phases for kind in ('call','call_exec','phase')if any(name(o)==cl and o['k']==kind for o in xs)]
 expansions=[o for o in ops if name(o)=='Artifacts.__itemsJSON'and o['k']=='call_exec'];assert len(expansions)==1
 expansion=expansions[0];addresses=sorted([o for o in children(expansion)if name(o)=='Address.container'and o['k']=='call_exec'],key=lambda o:o['s'])
 return {'profile_sha256':sha(p),'open_operations':len(h.get('open_ops',[])),'dropped_events':h['dropped_events'],'operation_count':len(ops),'phases':aggregates(ops),'expansion':{'duration_ms':(expansion['d']-expansion['s'])/1e6,'children':aggregates(children(expansion))},'addresses':[{'ordinal':i,'start_relative_to_expansion_ms':(o['s']-expansion['s'])/1e6,'duration_ms':(o['d']-o['s'])/1e6,'children':aggregates(children(o))}for i,o in enumerate(addresses)]}
def main():
 O.mkdir(exist_ok=True);rows=json.loads((R/'results.json').read_text());restoration=json.loads((R/'restoration.json').read_text())
 assert len(rows)==38 and all(r['correct']for r in rows)
 groups=[]
 for phase,flow in [('warm','checks'),('warm','artifacts'),('fresh-edit','checks')]:
  values={v:[r for r in rows if r['variant']==v and r['phase']==phase and r['flow']==flow]for v in ('baseline','candidate')}
  paired=[]
  for bb,cb in [(0,1),(3,2)]:
   b=[r for r in values['baseline']if r['block']==bb];c=[r for r in values['candidate']if r['block']==cb];assert len(b)==len(c)
   paired.extend((cv['seconds']-bv['seconds'])*1000 for bv,cv in zip(b,c))
  med={v:statistics.median(r['seconds']for r in vals)*1000 for v,vals in values.items()}
  groups.append({'phase':phase,'flow':flow,'milliseconds':{v:[r['seconds']*1000 for r in vals]for v,vals in values.items()},'median_ms':med,'median_change_percent':100*(med['candidate']/med['baseline']-1),'aligned_block_differences_ms':paired,'median_aligned_difference_ms':statistics.median(paired),'pairing':'Within aligned ABBA block positions; correlated observations, not independently randomized pairs.'})
 safe=[{k:r[k]for k in ('variant','flow','phase','block','profile','expected_failure','input_sha256','app_main_sha256','app_test_sha256','seconds','exit_code','correct','stdout_sha256','process_tree_user_seconds','process_tree_system_seconds','engine_written_bytes')}for r in rows]
 save(O/'runtime-numeric.json',{'groups':groups,'commands':safe,'restoration':restoration,'ordinary_excludes':['block primers','correctness-only calls','diagnostic primer','diagnostic profiled command'],'no_claims':['fresh-volume cold','Cloud','service up','SDK code generation'],'local_commands':38,'cloud_commands':0})
 profiles={}
 for d in sorted(R.glob('*-diagnostic-checks-*')):
  variant=d.name.rsplit('-',1)[1];profiles[variant]=profile(d/'run.wcprof.private')
 assert set(profiles)=={'baseline','candidate'};save(O/'profile-numeric.json',profiles)
 print(json.dumps({'groups':groups,'profiles':{k:{'operations':v['operation_count'],'expansion_ms':v['expansion']['duration_ms'],'addresses':len(v['addresses'])}for k,v in profiles.items()}},indent=2))
if __name__=='__main__':main()
