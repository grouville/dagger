def put(name,data):(OUT/name).write_text(json.dumps(data,indent=2)+'\n')

summary=[]
for phase in ['warm','fresh-edit']:
 for flow in dict.fromkeys(r['flow']for r in rows if r['phase']==phase):
  group={v:[r for r in rows if(r['phase'],r['flow'],r['variant'])==(phase,flow,v)]for v in ['baseline','candidate']}
  entry={'phase':phase,'flow':flow,'n_per_variant':3,'variants':{},'paired_candidate_minus_baseline_ms':[]}
  for v,g in group.items():
   entry['variants'][v]={'wall_ms':[r['seconds']*1000 for r in g],'median_wall_ms':median(g,'seconds')*1000,'median_cli_process_tree_cpu_ms':median(g,'process_tree_cpu_seconds')*1000,'median_engine_cpu_ms':median(g,'engine_cpu_seconds')*1000,'engine_written_bytes':[r['engine_written_bytes']for r in g],'file_visible_ms':[None if r['file_visible_seconds']is None else r['file_visible_seconds']*1000 for r in g],'visible_to_exit_ms':[None if r['file_visible_seconds']is None else (r['seconds']-r['file_visible_seconds'])*1000 for r in g]}
  for pair in range(3):
   a=next(r for r in group['baseline']if r['block']//2==pair);b=next(r for r in group['candidate']if r['block']//2==pair)
   entry['paired_candidate_minus_baseline_ms'].append((b['seconds']-a['seconds'])*1000)
  summary.append(entry)
