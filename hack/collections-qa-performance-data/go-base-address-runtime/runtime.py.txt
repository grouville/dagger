#!/usr/bin/env python3
"""Prepared 28-call LOCAL module comparison, gated on the preceding correctness proof.

Same Address/profiling engine, original CLI, ordinary init and retained volume.
Only the local Go module implementation and configured base/baseAddress differ.
Four ABBA blocks: exact-check primer, two warm checks listings, two first-seen
main.go comment listings, one artifact listing. Four final primer/profile calls
are separate. No execution, service, fresh-volume cold, or Cloud speed claim.
"""
from pathlib import Path
import argparse, hashlib, json, re, shutil, subprocess, sys, uuid
HERE=Path(__file__).resolve().parent
GATE=Path('/tmp/collections-perf/go-base-address-retained-v1')
LAB=Path('/tmp/collections-perf/engine-allocation-round2')
sys.path.insert(0,str(HERE))
from lifecycle import EngineLifecycle, ANSI, x, nav
CAP=28
FLOWS={'checks':['check','-l','--all'],'artifacts':['list','-a']}
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def normalize(raw):return [b' '.join(line.split())for line in ANSI.sub(b'',raw).splitlines()if line.strip()]
def main():
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('--run',action='store_true');a=p.parse_args()
 if not a.run:
  print(json.dumps({'execute':False,'local_cli_cap':CAP,'cloud_calls':0,'block_order':['control','candidate','candidate','control'],'block_primers':4,'warm_check_samples_per_arm':4,'unique_comment_samples_per_arm':4,'artifact_samples_per_arm':2,'separate_profile_primers':2,'profiles':2,'requires':'successful frozen 9-call correctness gate and reviewed frozen inputs','scope':'module/API comparison on one engine; no engine optimization attribution'}));return
 frozen=json.loads((HERE/'frozen-runtime-inputs.json').read_text())
 for path,want in frozen['source_sha256'].items():assert sha(path)==want,'frozen source changed: '+path
 summary=json.loads((GATE/'summary-v5.json').read_text())
 assert summary['validated_cli_commands']==9 and summary['synthetic_restored']and summary['greetings_authored_source_restored']and summary['original_greetings_untouched']
 assert sha(GATE/'summary-v5.json')==frozen['correctness_summary_sha256']
 gate_restored=json.loads((GATE/'results-v5/lifecycle-restoration.json').read_text())
 assert gate_restored['validated_commands']==9 and gate_restored['local_attempts']==9 and not gate_restored['cleanup_error_types']
 assert all(gate_restored[k]for k in ('temporary_container_removed','original_engine_stopped','original_engine_binary_untouched','original_init_binary_untouched','retained_volume_preserved'))
 control=Path(frozen['original_module']);candidate=Path(frozen['candidate_module'])
 original_app=LAB/'greetings';workspace=HERE/'greetings-v1';out=HERE/'results-v1'
 assert not workspace.exists()and not out.exists()and not(HERE/'summary-v1.json').exists()
 initial=json.loads((LAB/'withfile-v1/prepared.json').read_text())['input_sha256']
 assert all(sha(original_app/name)==want for name,want in initial.items())
 for name in initial:
  target=workspace/name;target.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(original_app/name,target)
 subprocess.run(['git','init','-q',str(workspace)],check=True,timeout=10)
 subprocess.run(['git','-C',str(workspace),'add','-f','.'],check=True,timeout=10)
 local=workspace/'.dagger/perf-go';local.mkdir(parents=True)
 (local/'dagger-module.toml').write_bytes((HERE/'module-manifest.toml').read_bytes())
 originals={n:(workspace/n).read_bytes()for n in ('dagger.toml','dagger.lock','main.go')}
 assert originals['dagger.toml'].count(b'source = "github.com/dagger/go@collections"')==1
 config=originals['dagger.toml'].replace(b'source = "github.com/dagger/go@collections"',b'source = ".dagger/perf-go"')
 needle=b'base = "dag://backend/go-test-base"';assert config.count(needle)==1
 configs={'control':config,'candidate':config.replace(needle,b'baseAddress = "dag://backend/go-test-base"')}
 expected_checks=normalize(Path('/home/dagger/dag/hack/collections-qa-performance-data/expected-checks.txt').read_bytes());assert len(expected_checks)==14
 expected={k:v for k,v in initial.items()if k!='dagger.lock'}
 goldens={};nonce=uuid.uuid4().hex;rows=[];active=None;lock_hashes=[]
 facts={'scope':'Module/API comparison; same Address engine and normal init in both arms. Original local Go source/base vs candidate local Go/baseAddress. Four correlated ABBA blocks, n4 checks and n4 first-seen comment listings per arm, n2 artifact observations per arm. Profiles excluded from ordinary timings. No cold, actual execution, service or Cloud latency claim.','cloud_commands':0,'cap':CAP}
 def put(name,body):
  path=workspace/name
  if not path.exists()or path.read_bytes()!=body:path.write_bytes(body)
  if name!='dagger.lock':expected[name]=sha(path)

 try:

  facts['ordinary_lock_write_hashes']=lock_hashes
  x.write(HERE/'summary-v1.json',facts)
  assert facts['copied_fixture_restored']and facts['original_fixture_untouched']
if __name__=='__main__':main()
