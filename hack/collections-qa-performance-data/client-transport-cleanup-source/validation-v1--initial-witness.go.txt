package client

import (
	"io"
	"net/http"
	"sync"
	"testing"
	"testing/synctest"

	"github.com/dagger/dagger/engine"
	"github.com/stretchr/testify/require"
)

func TestSharedHTTPTransportCloseOverlapsOtherCleanup(t *testing.T) {
	synctest.Test(t, func(t *testing.T) {
		c, _, peer := newSharedHTTPTestClient(t, func(w http.ResponseWriter, r *http.Request) {
			if r.URL.Path == engine.ShutdownEndpoint {
				w.WriteHeader(http.StatusNoContent)
				return
			}
			_, _ = w.Write([]byte("ok"))
		})
		resp := getSharedHTTP(t, c.httpClient, t.Context(), "/api")
		body, err := io.ReadAll(resp.Body)
		require.NoError(t, err)
		require.NoError(t, resp.Body.Close())
		require.Equal(t, "ok", string(body))

		peer.mu.Lock()
		connections := append([]chan struct{}(nil), peer.done...)
		peer.mu.Unlock()
		require.Len(t, connections, 1)
		physicalClosed := connections[0]
		allowOtherCleanup := make(chan struct{})
		releaseOtherCleanup := sync.OnceFunc(func() { close(allowOtherCleanup) })
		c.eg.Go(func() error { <-allowOtherCleanup; return nil })
		closed := make(chan struct{})
		var closeErr error
		go func() { closeErr = c.Close(); close(closed) }()
		t.Cleanup(func() { releaseOtherCleanup(); <-closed })

		synctest.Wait()
		select {
		case <-physicalClosed:
		default:
			t.Error("physical HTTP closure must not wait behind unrelated cleanup")
		}
		select {
		case <-closed:
			t.Error("Client.Close returned before other joined cleanup completed")
		default:
		}
		releaseOtherCleanup()
		<-closed
		require.NoError(t, closeErr)
		require.EqualValues(t, 1, peer.closes.Load())
	})
}
