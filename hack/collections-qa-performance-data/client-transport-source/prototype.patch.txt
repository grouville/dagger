+// The transport has one Client owner. HTTP wrappers retain separate request
+// headers and cancellation domains; neither wrapper owns a physical connection.
+// Tracking connections also covers active streams during failed initialization,
+// which CloseIdleConnections alone cannot close.
+type clientHTTPTransport struct {
+	transport *http2.Transport
+	lifetime  context.Context
+	cancel    context.CancelCauseFunc
+	dial      func(context.Context) (net.Conn, error)
+	mu        sync.Mutex
+	closed    bool
+	closeDone chan struct{}
+	pending   sync.WaitGroup
+	conns     map[*clientHTTPConn]struct{}
+}

+func newClientHTTPTransport(lifetime context.Context, dial func(context.Context) (net.Conn, error)) *clientHTTPTransport {
+	lifetime, cancel := context.WithCancelCause(lifetime)
+	t := &clientHTTPTransport{lifetime: lifetime, cancel: cancel, dial: dial, conns: make(map[*clientHTTPConn]struct{}), closeDone: make(chan struct{})}
+	t.transport = &http2.Transport{
+		AllowHTTP: true,
+		DialTLSContext: func(ctx context.Context, _, _ string, _ *tls.Config) (net.Conn, error) {
+			return t.connect(ctx)
+		},
+	}
+	return t
+}
+
+func (t *clientHTTPTransport) connect(ctx context.Context) (net.Conn, error) {
+	t.mu.Lock()
+	if t.closed {
+		t.mu.Unlock()
+		return nil, errors.New("client HTTP transport closed")
+	}
+	// Registration shares Close's lock, so no Add can race with its Wait.
+	t.pending.Add(1)
+	t.mu.Unlock()
+	defer t.pending.Done()
+	if t.lifetime.Err() != nil {
+		return nil, context.Cause(t.lifetime)
+	}
+
+	// Cancellation of the opening request must still interrupt establishment.
+	// Once handed to the pool, only the Client owns the connection lifetime.
+	dialCtx, cancel := context.WithCancelCause(context.WithoutCancel(ctx))
+	stopLifetime := context.AfterFunc(t.lifetime, func() { cancel(context.Cause(t.lifetime)) })
+	var dialMu sync.Mutex
+	dialing := true
+	stopRequest := context.AfterFunc(ctx, func() {
+		dialMu.Lock()
+		defer dialMu.Unlock()
+		if dialing {
+			cancel(context.Cause(ctx))
+		}
+	})
+	conn, err := t.dial(dialCtx)
+	dialMu.Lock()
+	dialing = false
+	dialMu.Unlock()
+	stopRequest()
+	if err == nil && dialCtx.Err() != nil {
+		err = context.Cause(dialCtx)
+		_ = conn.Close()
+	}
+	if err != nil {
+		stopLifetime()
+		cancel(err)
+		return nil, err
+	}
+
+	owned := &clientHTTPConn{Conn: conn, owner: t, release: func() {
+		stopLifetime()
+		cancel(context.Canceled)
+	}}
+	t.mu.Lock()
+	if t.closed {
+		t.mu.Unlock()
+		_ = owned.Close()
+		return nil, errors.New("client HTTP transport closed")
+	}
+	t.conns[owned] = struct{}{}
+	t.mu.Unlock()
+	return owned, nil
+}
+
+func (t *clientHTTPTransport) Close() {
+	t.mu.Lock()
+	if t.closed {
+		done := t.closeDone
+		t.mu.Unlock()
+		<-done
+		return
+	}
+	t.closed = true
+	conns := make([]*clientHTTPConn, 0, len(t.conns))
+	for conn := range t.conns {
+		conns = append(conns, conn)
+	}
+	t.mu.Unlock()
+	t.cancel(errors.New("client HTTP transport closed"))
+	for _, conn := range conns {
+		_ = conn.Close()
+	}
+	// A connector that returns after cancellation still hands back an owned
+	// connection: connect closes it before decrementing pending. In particular,
+	// Docker command startup ignores context cancellation until it returns.
+	t.pending.Wait()
+	t.transport.CloseIdleConnections()
+	close(t.closeDone)
+}
+
+type clientHTTPConn struct {
+	net.Conn
+	owner   *clientHTTPTransport
+	release func()
+	once    sync.Once
+	err     error
+}
+
+func (c *clientHTTPConn) Close() error {
+	c.once.Do(func() {
+		c.err = c.Conn.Close()
+		c.release()
+		c.owner.mu.Lock()
+		delete(c.owner.conns, c)
+		c.owner.mu.Unlock()
+	})
+	return c.err
+}

+func TestSharedHTTPTransportCloseJoinsPendingDial(t *testing.T) {
+	synctest.Test(t, func(t *testing.T) {
+		started := make(chan struct{})
+		canceled := make(chan struct{})
+		allowReturn := make(chan struct{})
+		closeStarted := make(chan struct{})
+		allowPhysicalClose := make(chan struct{})
+		var returnOnce, closeOnce sync.Once
+		releaseDial := func() { returnOnce.Do(func() { close(allowReturn) }) }
+		releaseClose := func() { closeOnce.Do(func() { close(allowPhysicalClose) }) }
+		defer releaseDial()
+		defer releaseClose()
+		owner := newClientHTTPTransport(t.Context(), func(ctx context.Context) (net.Conn, error) {
+			close(started)
+			<-ctx.Done()
+			close(canceled)
+			<-allowReturn // Model command startup that ignores cancellation.
+			left, right := net.Pipe()
+			_ = right.Close()
+			return &sharedHTTPGatedCloseConn{Conn: left, started: closeStarted, allow: allowPhysicalClose}, nil
+		})
+		dialDone := make(chan struct{})
+		go func() { _, _ = owner.connect(t.Context()); close(dialDone) }()
+		<-started
+		firstDone := make(chan struct{})
+		secondDone := make(chan struct{})
+		go func() { owner.Close(); close(firstDone) }()
+		<-canceled
+		go func() { owner.Close(); close(secondDone) }()
+		synctest.Wait()
+		select {
+		case <-firstDone:
+			t.Error("first Close returned before pending dial completed")
+		default:
+		}
+		select {
+		case <-secondDone:
+			t.Error("concurrent Close did not join the owner")
+		default:
+		}
+		releaseDial()
+		<-closeStarted
+		synctest.Wait()
+		select {
+		case <-firstDone:
+			t.Error("Close returned before physical connection cleanup")
+		default:
+		}
+		select {
+		case <-secondDone:
+			t.Error("concurrent Close returned before physical cleanup")
+		default:
+		}
+		releaseClose()
+		<-dialDone
+		<-firstDone
+		<-secondDone
+		owner.Close()
+	})
+}
+
+type sharedHTTPGatedCloseConn struct {
+	net.Conn
+	started chan struct{}
+	allow   <-chan struct{}
+}
+
+func (c *sharedHTTPGatedCloseConn) Close() error {
+	close(c.started)
+	<-c.allow
+	return c.Conn.Close()
+}
