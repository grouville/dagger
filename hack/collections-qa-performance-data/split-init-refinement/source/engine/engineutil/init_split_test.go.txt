package engineutil

import (
	"context"
	"reflect"
	"testing"

	"github.com/dagger/dagger/engine"
	"github.com/dagger/dagger/engine/distconsts"
	"github.com/dagger/dagger/internal/buildkit/executor"
)

func TestSplitInitMounts(t *testing.T) {
	for _, tc := range []struct {
		name        string
		nested      bool
		noInit      bool
		env         []string
		wantSession bool
	}{
		{name: "ordinary"},
		{name: "nested", nested: true, wantSession: true},
		{name: "explicit token", env: []string{"DAGGER_SESSION_TOKEN=test"}, wantSession: true},
		{name: "empty token", env: []string{"DAGGER_SESSION_TOKEN="}, wantSession: true},
		{name: "similar key", env: []string{"DAGGER_SESSION_TOKEN_OTHER=test"}},
		{name: "not assignment", env: []string{"DAGGER_SESSION_TOKEN"}},
		{name: "explicit token no init", env: []string{"DAGGER_SESSION_TOKEN=test"}, noInit: true},
		{name: "no init", noInit: true},
		{name: "no nested init", nested: true, noInit: true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			originalArgs := []string{"sh", "-c", "exit 0"}
			state := &execState{
				procInfo: &executor.ProcessInfo{Meta: executor.Meta{Args: originalArgs, Env: tc.env}},
				execMD:   &ExecutionMetadata{NoInit: tc.noInit},
			}
			if tc.nested {
				state.nestedClientMetadata = &engine.ClientMetadata{ClientID: "test-only-nested"}
			}
			if err := (&Client{}).injectInit(context.Background(), state); err != nil {
				t.Fatal(err)
			}
			if tc.noInit {
				if len(state.mounts) != 0 || !reflect.DeepEqual(state.procInfo.Meta.Args, originalArgs) {
					t.Fatal("NoInit must retain original argv and mount list")
				}
				return
			}
			if !reflect.DeepEqual(state.procInfo.Meta.Args, append([]string{"/.init"}, originalArgs...)) {
				t.Fatalf("argv changed: %q", state.procInfo.Meta.Args)
			}
			wants := []struct{ source, destination string }{{distconsts.DaggerInitLitePath, "/.init"}}
			if tc.wantSession {
				wants = append(wants, struct{ source, destination string }{distconsts.DaggerInitPath, distconsts.DaggerSessionContainerPath})
			}
			if len(state.mounts) != len(wants) {
				t.Fatalf("want %d mounts, got %d", len(wants), len(state.mounts))
			}
			for i, want := range wants {
				got := state.mounts[i]
				bind, ok := got.Src.(hostBindMount)
				if !ok || bind.srcPath != want.source || got.Dest != want.destination || !got.Readonly {
					t.Fatalf("mount %d differs from explicit read-only contract", i)
				}
			}
		})
	}
}
