package apply

import (
	"bytes"
	"errors"
	"io"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

type progressTestReader struct {
	*bytes.Reader
	closes atomic.Int64
	read   func([]byte) (int, error)
	close  func() error
}

func (r *progressTestReader) Close() error {
	r.closes.Add(1)
	if r.close != nil {
		return r.close()
	}
	return nil
}

// content.NewReader's optional streaming interface avoids unrelated section
// reader buffering in these reader-lifecycle tests.
type progressTestSource struct{ source *progressTestReader }

func (r progressTestSource) Reader() io.Reader { return r.source }
func (r progressTestSource) ReadAt(p []byte, off int64) (int, error) {
	return r.source.Reader.ReadAt(p, off)
}
func (r progressTestSource) Size() int64  { return r.source.Size() }
func (r progressTestSource) Close() error { return r.source.Close() }
func (r *progressTestReader) Read(p []byte) (int, error) {
	if r.read != nil {
		return r.read(p)
	}
	return r.Reader.Read(p)
}

func TestProgressReaderCloseIsTerminal(t *testing.T) {
	source := &progressTestReader{Reader: bytes.NewReader([]byte("payload"))}
	var progress []int64
	r := newProgressReader(progressTestSource{source}, func(n int64) { progress = append(progress, n) })
	if _, err := io.Copy(io.Discard, r); err != nil {
		t.Fatal(err)
	}
	if err := r.Close(); err != nil {
		t.Fatal(err)
	}
	if len(progress) == 0 || progress[0] != 0 || progress[len(progress)-1] != 7 {
		t.Fatalf("expected initial and converged final progress: %v", progress)
	}
	nBefore := len(progress)
	if _, err := r.Read(make([]byte, 1)); !errors.Is(err, io.ErrClosedPipe) {
		t.Errorf("read after close = %v, want io.ErrClosedPipe", err)
	}
	if err := r.Close(); err != nil {
		t.Fatal(err)
	}
	if len(progress) != nBefore || source.closes.Load() != 1 {
		t.Fatalf("close must be terminal/idempotent: events %d -> %d, source closes %d", nBefore, len(progress), source.closes.Load())
	}
}

func TestProgressReaderConcurrentClose(t *testing.T) {
	for range 20 {
		source := &progressTestReader{Reader: bytes.NewReader(bytes.Repeat([]byte("x"), 1<<18))}
		var mu sync.Mutex
		var progress []int64
		r := newProgressReader(progressTestSource{source}, func(n int64) {
			mu.Lock()
			progress = append(progress, n)
			mu.Unlock()
		})
		started := make(chan struct{})
		done := make(chan struct{})
		go func() {
			defer close(done)
			b := make([]byte, 7)
			_, _ = r.Read(b)
			close(started)
			for {
				if _, err := r.Read(b); err != nil {
					return
				}
			}
		}()
		<-started
		if err := r.Close(); err != nil {
			t.Fatal(err)
		}
		mu.Lock()
		atClose := len(progress)
		mu.Unlock()
		<-done
		mu.Lock()
		after := len(progress)
		mu.Unlock()
		if after != atClose {
			t.Fatalf("progress emitted after Close returned: %d -> %d", atClose, after)
		}
	}
}

func TestProgressReaderCloseDoesNotJoinBorrowedRead(t *testing.T) {
	entered, release := make(chan struct{}), make(chan struct{})
	var releaseOnce sync.Once
	unblock := func() { releaseOnce.Do(func() { close(release) }) }
	defer unblock()
	source := &progressTestReader{Reader: bytes.NewReader([]byte("x"))}
	source.read = func(p []byte) (int, error) {
		close(entered)
		<-release
		p[0] = 'x'
		return 1, io.EOF
	}
	r := newProgressReader(progressTestSource{source}, func(int64) {})
	readDone := make(chan struct{})
	go func() {
		defer close(readDone)
		_, _ = r.Read(make([]byte, 1))
	}()
	<-entered
	closeDone := make(chan struct{})
	go func() { _ = r.Close(); close(closeDone) }()
	select {
	case <-closeDone:
	case <-time.After(time.Second):
		unblock()
		<-closeDone
		<-readDone
		t.Fatal("Close must not assume arbitrary borrowed input is interruptible")
	}
	unblock()
	<-readDone
}

func TestProgressReaderConcurrentClosePreservesError(t *testing.T) {
	entered, release := make(chan struct{}), make(chan struct{})
	var unblockOnce sync.Once
	unblock := func() { unblockOnce.Do(func() { close(release) }) }
	defer unblock()
	sentinel := errors.New("source close failed")
	source := &progressTestReader{Reader: bytes.NewReader([]byte("payload"))}
	source.close = func() error { close(entered); <-release; return sentinel }
	var callbacks atomic.Int64
	r := newProgressReader(progressTestSource{source}, func(int64) { callbacks.Add(1) })
	closed := make(chan error, 2)
	go func() { closed <- r.Close() }()
	<-entered
	started := make(chan struct{})
	go func() { close(started); closed <- r.Close() }()
	<-started
	if callbacks.Load() != 0 {
		t.Error("terminal progress preceded source close completion")
	}
	unblock()
	for range 2 {
		if err := <-closed; !errors.Is(err, sentinel) {
			t.Errorf("Close lost source error: %v", err)
		}
	}
	if source.closes.Load() != 1 || callbacks.Load() != 1 {
		t.Fatalf("source close/terminal callback must occur once: %d/%d", source.closes.Load(), callbacks.Load())
	}
	if _, err := r.Read(make([]byte, 1)); !errors.Is(err, io.ErrClosedPipe) || callbacks.Load() != 1 {
		t.Fatalf("read after Close: err=%v callbacks=%d", err, callbacks.Load())
	}
}

func TestProgressReaderCallbackPanicDoesNotLockCleanup(t *testing.T) {
	source := &progressTestReader{Reader: bytes.NewReader([]byte("payload"))}
	first := true
	r := newProgressReader(progressTestSource{source}, func(int64) {
		if first {
			first = false
			panic("progress callback panic")
		}
	})
	func() {
		defer func() {
			if got := recover(); got != "progress callback panic" {
				t.Errorf("unexpected recovered callback result: %v", got)
			}
		}()
		_, _ = r.Read(make([]byte, 1))
	}()
	done := make(chan error, 1)
	go func() { done <- r.Close() }()
	select {
	case err := <-done:
		if err != nil {
			t.Fatal(err)
		}
	case <-time.After(time.Second):
		t.Fatal("callback panic left the progress mutex locked during cleanup")
	}
}
